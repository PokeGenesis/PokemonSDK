    - Pattern :
      ```csharp
      public sealed class NarrationQueue : IDisposable
      {
          private readonly Channel<string> _channel = Channel.CreateUnbounded<string>();
          private readonly CancellationTokenSource _cts = new();
          private readonly Task _processingTask;
          private readonly INarrationPlugin _plugin;

          public NarrationQueue(INarrationPlugin plugin)
          {
              _plugin = plugin;
              _processingTask = Task.Run(() => ProcessAsync(_cts.Token));
          }

          public void Enqueue(string text) => _channel.Writer.TryWrite(text);

          public void Stop()
          {
              _cts.Cancel();
              _channel.Writer.TryComplete();
          }

          private async Task ProcessAsync(CancellationToken ct)
          {
              await foreach (var text in _channel.Reader.ReadAllAsync(ct))
                  await _plugin.SpeakAsync(text, ct);
          }

          public void Dispose() { Stop(); _cts.Dispose(); }
      }
      ```

    **PiperTtsPlugin.cs** (reviewer concerns #1 + #2) :
    - Constructor : `(string piperPath, string voiceModel)`. Stocker paths.
    - `IsSupported` : `File.Exists(piperPath) || IsOnPath(piperPath)` — méthode helper `IsOnPath` check via `which` ou `where`.
    - `IsSpeaking` : `_currentProcess != null && !_currentProcess.HasExited`
    - `SpeakAsync(string text, CancellationToken ct)` :
      ```
      if (!IsSupported) return;
      var inputFile = Path.GetTempFileName() + ".txt";
      var outputFile = Path.GetTempFileName() + ".wav";
      try {
          File.WriteAllText(inputFile, text);
          // 1. Lancer piper
          using var piperProc = Process.Start(new ProcessStartInfo(piperPath) {
              Arguments = $"--input \"{inputFile}\" --output_file \"{outputFile}\" --model \"{voiceModel}\"",
              RedirectStandardOutput = true,
              RedirectStandardError = true
          })!;
          await piperProc.WaitForExitAsync(ct);
          // 2. Vérifier exit code AVANT de lancer aplay (reviewer concern #1)
          if (piperProc.ExitCode != 0) return;  // ou throw si strict
          if (!File.Exists(outputFile)) return;
          // 3. Lancer player audio
          var player = DeterminePlayer();  // "aplay" Linux, "mplayer" fallback
          if (player == null) return;
          using var audioProc = Process.Start(new ProcessStartInfo(player) {
              Arguments = $"\"{outputFile}\""
          })!;
          await audioProc.WaitForExitAsync(ct);
      } finally {
          // Cleanup temp files (reviewer concern #2)
          try { File.Delete(inputFile); } catch { }
          try { File.Delete(outputFile); } catch { }
      }
      ```
    - `Stop()` : kill `_currentProcess` si en cours, `_cts.Cancel()`
    - `DeterminePlayer()` : check `aplay` en premier, puis `paplay`, puis `mplayer`, sinon null.
