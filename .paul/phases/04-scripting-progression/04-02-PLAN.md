    ```csharp
    public class BadgeApi
    {
        private GameState _state;
        public BadgeApi(GameState initial) => _state = initial;
        public void AwardBadge(string id) => _state = _state.WithFlag($"badge_{id}", true);
        public bool HasBadge(string id) => _state.GetFlag<bool>($"badge_{id}");
        public GameState GetState() => _state;
    }
    ```
    Créer src/SDK.Scripting/Bindings/NpcInteractionRunner.cs :
    ```csharp
    public static class NpcInteractionRunner
    {
        public static GameState Run(IScriptEngine engine, GameState state, string luaScript)
        {
            var api = new BadgeApi(state);
            engine.RegisterApi("badges", api);
            engine.Execute(luaScript);
            return api.GetState();
        }
    }
    ```
