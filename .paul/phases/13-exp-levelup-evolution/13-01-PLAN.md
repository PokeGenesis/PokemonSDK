    Après le log "X fainted!", si `_expFormula != null`, appeler `AwardExp(ref newState)` :

    ```csharp
    private BattleState AwardExp(BattleState state)
    {
        var formula = _expFormula!;
        var player = state.Player;
        var opponent = state.Opponent;

        int gained = formula.CalcExpGain(opponent.BaseExpYield, opponent.Level, false);
        int newExp = player.CurrentExp + gained;

        var log = state.Log.ToList();
        log.Add($"{player.Nickname} gained {gained} EXP!");

        int newLevel = player.Level;
        int atk = player.Attack, def = player.Defense;
        int spa = player.SpecialAttack, spd = player.SpecialDefense, spe = player.Speed;
        int maxHp = player.MaxHp;

        while (newLevel < 100 && newExp >= formula.ExpThreshold(newLevel + 1, player.GrowthRate))
        {
            int oldLevel = newLevel;
            newLevel++;

            double scale = (newLevel + 5.0) / (oldLevel + 5.0);
            atk    = (int)(atk    * scale);
            def    = (int)(def    * scale);
            spa    = (int)(spa    * scale);
            spd    = (int)(spd    * scale);
            spe    = (int)(spe    * scale);
            maxHp  = (int)(maxHp  * scale);

            log.Add($"{player.Nickname} grew to level {newLevel}!");

            if (player.FullLearnset != null)
                foreach (var (learnLevel, move) in player.FullLearnset)
                    if (learnLevel == newLevel)
                        log.Add($"{player.Nickname} learned {move.Identifier}!");

            var updatedPlayer = player with
            {
                Level = newLevel, CurrentExp = newExp,
                Attack = atk, Defense = def,
                SpecialAttack = spa, SpecialDefense = spd, Speed = spe,
                MaxHp = maxHp,
            };
            _plugins?.NotifyLevelUp(updatedPlayer, oldLevel, newLevel);
            player = updatedPlayer;
        }

        player = player with { CurrentExp = newExp };
        return state with { Player = player, Log = log };
    }
    ```

    using FluentAssertions;
    using SDK.Battle.Formulas;

    namespace SDK.Battle.Tests;

    public sealed class ExpFormulaTests
    {
        // Gen1 — CalcExpGain
        [Theory]
        [InlineData(64, 25, false, 228)] // (1.0 * 64 * 25 / 7)
        [InlineData(64, 25, true,  342)] // (1.5 * 64 * 25 / 7)
        [InlineData(100, 50, false, 714)] // (1.0 * 100 * 50 / 7)
        public void Gen1_CalcExpGain_MatchesFormula(int baseYield, int opponentLevel, bool trainerBattle, int expected)
        {
            var formula = new Gen1ExpFormula();
            formula.CalcExpGain(baseYield, opponentLevel, trainerBattle).Should().Be(expected);
        }

        // Gen1 — ExpThreshold par GrowthRate
        [Theory]
        [InlineData(10, GrowthRate.MediumFast, 1000)]   // 10^3
        [InlineData(10, GrowthRate.Fast,        800)]   // 4*10^3/5
        [InlineData(10, GrowthRate.Slow,        1250)]  // 5*10^3/4
        public void Gen1_ExpThreshold_MatchesGrowthRate(int level, GrowthRate rate, int expected)
        {
            var formula = new Gen1ExpFormula();
            formula.ExpThreshold(level, rate).Should().Be(expected);
        }

        // Gen5 — CalcExpGain (non-linéaire)
        [Theory]
        [InlineData(64, 25, false)] // résultat > 0 et < Gen1 pour niveaux bas
        [InlineData(64, 50, true)]
        public void Gen5_CalcExpGain_ReturnsPositive(int baseYield, int opponentLevel, bool trainerBattle)
        {
            var formula = new Gen5ExpFormula();
            formula.CalcExpGain(baseYield, opponentLevel, trainerBattle).Should().BeGreaterThan(0);
        }

        // Gen5 — ExpThreshold identique à Gen1 (même courbes)
        [Theory]
        [InlineData(10, GrowthRate.MediumFast, 1000)]
        [InlineData(20, GrowthRate.Slow,        10000)] // 5*20^3/4
        public void Gen5_ExpThreshold_SameAsGen1(int level, GrowthRate rate, int expected)
        {
            var formula = new Gen5ExpFormula();
            formula.ExpThreshold(level, rate).Should().Be(expected);
        }

        // ExpThreshold croissant avec le niveau
        [Fact]
        public void ExpThreshold_IsStrictlyIncreasing_ForMediumFast()
        {
            var formula = new Gen1ExpFormula();
            for (int level = 2; level <= 99; level++)
                formula.ExpThreshold(level, GrowthRate.MediumFast)
                    .Should().BeGreaterThan(formula.ExpThreshold(level - 1, GrowthRate.MediumFast));
        }
    }
