    | 175 | Togepi      | Togepi      | Togepi   | Togepi      | トゲピー     |

    Utiliser une boucle sur les locales pour éviter la duplication :
    ```csharp
    var data = new (int id, string en, string fr, string de, string es, string ja)[]
    {
        (  1, "Bulbasaur", "Bulbizarre", "Bisasam", "Bulbasaur", "フシギダネ"),
        ( 25, "Pikachu",   "Pikachu",   "Pikachu", "Pikachu",   "ピカチュウ"),
        (175, "Togepi",    "Togepi",    "Togepi",  "Togepi",    "トゲピー"),

    foreach (var (id, en, fr, de, es, ja) in data)
    ```

  <action>

    Utiliser une fixture locale (`new SqliteTestFixture()` par test — SeedAll mute l'état).

    Test unique couvrant la Phase 1 goal :

    ```csharp
    [Fact]
    public void Phase1_Can_Persist_Query_By_Generation_And_Read_In_5_Locales()
    {
        // Arrange — seed complet
        using var fixture = new SqliteTestFixture();
        using (var ctx = fixture.CreateContext())
            DataSeeder.SeedAll(ctx);

        using var ctx2 = fixture.CreateContext();

        // --- Phase 1 goal: filtre génération ---
        var gen1Species = ctx2.GetSpeciesByGeneration(1).Select(s => s.Identifier).ToList();
        gen1Species.Should().Contain("bulbasaur");
        gen1Species.Should().Contain("pikachu");
        gen1Species.Should().NotContain("togepi"); // gen2 — exclu

        // --- Phase 1 goal: lecture en 5 locales ---
        using var ctx3 = fixture.CreateContext();
        var locales = new[] { "en", "fr", "de", "es", "ja" };
        foreach (var locale in locales)
        {
            var name = ctx3.GetTranslation("PokemonSpecies", 1, locale, "name");
            name.Should().NotBeNullOrEmpty($"Bulbasaur doit avoir un nom en {locale}");
        }

        // fr spécifique — valeur exacte
        ctx3.GetTranslation("PokemonSpecies", 1, "fr", "name").Should().Be("Bulbizarre");
        // ja spécifique — valeur exacte
        ctx3.GetTranslation("PokemonSpecies", 25, "ja", "name").Should().Be("ピカチュウ");

        // --- Vérification 5 locales distinctes en DB ---
        using var ctx4 = fixture.CreateContext();
        var distinctLocales = ctx4.Translations
            .Where(t => t.EntityType == "PokemonSpecies")
            .Select(t => t.Locale)
            .Distinct()
            .ToList();
        distinctLocales.Should().HaveCount(5);
        distinctLocales.Should().BeEquivalentTo(locales);
    }
    ```
  </action>

    ```csharp
    public void AllProjects_TargetNet10()
    {
        var root = Path.GetFullPath(
            Path.Combine(AppContext.BaseDirectory, "..", "..", "..", "..", ".."));

        var csprojFiles = Directory.GetFiles(root, "*.csproj", SearchOption.AllDirectories)
            .Where(f => !f.Contains(Path.DirectorySeparatorChar + "obj" + Path.DirectorySeparatorChar))
            .ToList();

        csprojFiles.Should().NotBeEmpty("le repo doit contenir des .csproj");

        foreach (var f in csprojFiles)
        {
            var doc = System.Xml.Linq.XDocument.Load(f);
            var tf = doc.Descendants("TargetFramework").FirstOrDefault()?.Value;
            tf.Should().Be("net10.0",
                $"{Path.GetFileName(f)} doit cibler net10.0 (D-01)");
        }
    }
    ```

    ```csharp
    [Fact]
    public void SourceFiles_ContainNoHardcodedWindowsPaths()
    {
        var root = Path.GetFullPath(
            Path.Combine(AppContext.BaseDirectory, "..", "..", "..", "..", ".."));

        var srcDir = Path.Combine(root, "src");
        var csFiles = Directory.GetFiles(srcDir, "*.cs", SearchOption.AllDirectories)
            .Where(f => !f.Contains(Path.DirectorySeparatorChar + "obj" + Path.DirectorySeparatorChar))
    ```
