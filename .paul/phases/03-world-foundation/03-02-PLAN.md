      namespace SDK.Core.Services;
      using SDK.Core.Enums;
      using SDK.Core.Interfaces;
      public class RealTimeClock(Func<DateTime>? clock = null) : IRealTimeClock
      {
          private readonly Func<DateTime> _clock = clock ?? (() => DateTime.UtcNow);
          public TimeOfDay GetTimeOfDay() => MapHour(_clock().Hour);
          internal static TimeOfDay MapHour(int hour) => hour switch
          {
              >= 6 and < 11  => TimeOfDay.Morning,
              >= 11 and < 17 => TimeOfDay.Day,
              >= 17 and < 21 => TimeOfDay.Evening,
              _              => TimeOfDay.Night,
          };
      }
      Note : Func<DateTime> injectable → testable. MapHour internal pour partage avec GameTimeClock.

    Créer src/SDK.Core/Services/GameTimeClock.cs :
      namespace SDK.Core.Services;
      using SDK.Core.Enums;
      using SDK.Core.Interfaces;
      public class GameTimeClock : IGameClock
      {
          private TimeSpan _elapsed = TimeSpan.Zero;
          public float Speed { get; set; } = 1f / 60f;  // 1:1 par défaut
          public TimeSpan GameElapsed => _elapsed;
          public void Update(TimeSpan realDelta)
              => _elapsed += TimeSpan.FromMinutes(realDelta.TotalSeconds * Speed);
          public void SetGameTime(TimeSpan elapsed) => _elapsed = elapsed;
          public TimeOfDay GetTimeOfDay()
          {
              int gameHour = (int)(_elapsed.TotalHours % 24);
              return RealTimeClock.MapHour(gameHour);
          }
      }
      Note : _elapsed = time game accumulé. gameHour = total heures modulo 24.
      Speed = game-minutes-par-real-second → Update(1s) avec Speed=1f → +1 min game.
      Speed=60f → 1s réelle = 1h game → jour complet en 24s.
      SetGameTime() : save/load restaure le total de temps écoulé.
      Éviter : threading, locks — mis à jour depuis le thread principal MonoGame uniquement.

              clock.GetTimeOfDay().Should().Be(expected);
          }

      public class GameTimeClockTests
      {
          [Fact]
          public void Update_AccumulatesGameTime()
          {
              var sut = new GameTimeClock { Speed = 1f }; // 1 real-sec = 1 game-min
              sut.Update(TimeSpan.FromSeconds(60));
              sut.GameElapsed.Should().BeCloseTo(TimeSpan.FromMinutes(60), TimeSpan.FromSeconds(1));
          }

          [Fact]
          public void GetTimeOfDay_MorningAfter6GameHours()
          {
              var sut = new GameTimeClock { Speed = 60f }; // 1s réelle = 1h game
              sut.Update(TimeSpan.FromSeconds(6));         // → 6h game
              sut.GetTimeOfDay().Should().Be(TimeOfDay.Morning);
          }

          [Fact]
          public void GetTimeOfDay_WrapsAt24Hours()
          {
              var sut = new GameTimeClock { Speed = 60f };
              sut.Update(TimeSpan.FromSeconds(30)); // 30h game → wrappe à 6h → Morning
              sut.GetTimeOfDay().Should().Be(TimeOfDay.Morning);
          }

          [Fact]
          public void SetGameTime_RestoresElapsed()
          {
              var sut = new GameTimeClock();
              var saved = TimeSpan.FromHours(14);
              sut.SetGameTime(saved);
              sut.GameElapsed.Should().Be(saved);
              sut.GetTimeOfDay().Should().Be(TimeOfDay.Day); // 14h = Day
          }

          [Fact]
          public void Speed_ConfigurableAtRuntime()

              sut.Update(TimeSpan.FromSeconds(60)); // +120 game-min
