    **publish-nuget.yml — réécriture complète :**

    ```yaml
    name: Publish NuGet

    on:
      push:
        tags: ['v*.*.*']

    jobs:
      publish:
        runs-on: ubuntu-latest

        steps:
          - uses: actions/checkout@v4

          - name: Setup .NET 10
            uses: actions/setup-dotnet@v4
            with:
              dotnet-version: '10.0.x'

          - name: Restore tools (MGCB)
            run: dotnet tool restore

          - name: Restore packages
            run: dotnet restore PokemonSDK.slnx

          - name: Build Release
            env:
              SIXLABORS_LICENSE_KEY: ${{ secrets.SIXLABORS_LICENSE_KEY }}
            run: |
              dotnet build PokemonSDK.slnx \
                --configuration Release \
                --no-restore \
                -p:SixLaborsLicenseKey="$SIXLABORS_LICENSE_KEY"

          - name: Run all tests
            run: dotnet test PokemonSDK.slnx --configuration Release --no-build

          - name: Pack 7 packages PokeForge.SDK.*
            run: |
              dotnet pack PokemonSDK.slnx \
                --configuration Release \
                --no-build \
                -o ./nupkg

          - name: Push vers NuGet.org
            env:
              NUGET_API_KEY: ${{ secrets.NUGET_API_KEY }}
            run: |
              if [ -z "$NUGET_API_KEY" ]; then
                echo "NUGET_API_KEY non configuré — push ignoré"
                exit 0
              fi
              dotnet nuget push ./nupkg/*.nupkg \
                --api-key "$NUGET_API_KEY" \
                --source https://api.nuget.org/v3/index.json \
                --skip-duplicate

          - name: Upload nupkg artifacts
            uses: actions/upload-artifact@v4
            with:
              name: nuget-packages-${{ github.ref_name }}
              path: ./nupkg
    ```

    **ci.yml — ajouter SIXLABORS_LICENSE_KEY au step Build uniquement :**

    Trouver le step `Build` (qui contient `dotnet build PokemonSDK.slnx --no-restore --configuration Release`) et le modifier pour injecter le secret :

    ```yaml
          - name: Build
            env:
              SIXLABORS_LICENSE_KEY: ${{ secrets.SIXLABORS_LICENSE_KEY }}
            run: dotnet build PokemonSDK.slnx --no-restore --configuration Release -p:SixLaborsLicenseKey="$SIXLABORS_LICENSE_KEY"
    ```

    Le step `Restore` et `Test + Coverage` ne nécessitent pas le secret (pas de compilation, DLLs déjà buildés).

    **Ne pas modifier** les autres steps de ci.yml (Restore, Test+Coverage, validate SDK.Core deps, headless smoke test).
