    ```csharp
    namespace SDK.Battle.Plugins;

    using SDK.Core.Interfaces;
    using SDK.Core.ValueObjects;

    public sealed class PluginRegistry
    {
        private readonly List<IBattlePlugin> _plugins = new();

        public void Register(IBattlePlugin plugin)
        {
            if (_plugins.Any(p => p.Name == plugin.Name))
                throw new InvalidOperationException(
                    $"Plugin '{plugin.Name}' is already registered.");
            _plugins.Add(plugin);
        }

        public void Unregister(string name) =>
            _plugins.RemoveAll(p => p.Name == name);

        public bool IsRegistered(string name) =>
            _plugins.Any(p => p.Name == name);

        // Observers (void)
        public void NotifyBattleStart(BattleState state)    => _plugins.ForEach(p => p.OnBattleStart(state));
        public void NotifyTurnStart(BattleState state)      => _plugins.ForEach(p => p.OnTurnStart(state));
        public void NotifyTurnEnd(BattleState state)        => _plugins.ForEach(p => p.OnTurnEnd(state));
        public void NotifyBattleEnd(BattleState s, BattleResult r) => _plugins.ForEach(p => p.OnBattleEnd(s, r));
        public void NotifyFainted(BattleState s, BattlePokemon f)  => _plugins.ForEach(p => p.OnPokemonFainted(s, f));
        public void NotifyCaught(BattleState s, BattlePokemon c, string zone) =>
            _plugins.ForEach(p => p.OnPokemonCaught(s, c, zone));

        // Chain state — null retour d'un plugin = pas de modif, passe à suivant
        public BattleState ApplyBeforeMove(BattleState state, BattleAction action)
        {
            foreach (var plugin in _plugins)
                if (plugin.OnBeforeMove(state, action) is { } next)
                    state = next;
            return state;
        }

        public BattleState ApplyBeforeDamage(BattleState state, DamageResult damage)
        {
            foreach (var plugin in _plugins)
                if (plugin.OnBeforeDamage(state, damage) is { } next)
                    state = next;
            return state;
        }
    }
    ```
