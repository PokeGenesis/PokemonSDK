```
IPlugin (SDK.Core.Interfaces)      ← base : string Name uniquement

  IEncounterPlugin : IPlugin       ← Phase 6 (OnWildEncounter, OnPokemonCaught, OnZoneEnter)

- NuzlockePlugin : IBattlePlugin (+ IEncounterPlugin Phase 6)

    public sealed class PluginRegistry
    {
        private readonly List<IPlugin> _plugins = [];

        public void Register(IPlugin plugin)
        {
            if (_plugins.Any(p => p.Name == plugin.Name))
                throw new InvalidOperationException($"Plugin '{plugin.Name}' already registered.");
            _plugins.Add(plugin);
        }

        public void Unregister(string name) => _plugins.RemoveAll(p => p.Name == name);

        public bool IsRegistered(string name) => _plugins.Any(p => p.Name == name);

        public void NotifyBattleStart(BattleState state)
        {
            foreach (var p in _plugins.OfType<IBattlePlugin>()) p.OnBattleStart(state);
        }

        public void NotifyTurnStart(BattleState state)
        {
            foreach (var p in _plugins.OfType<IBattlePlugin>()) p.OnTurnStart(state);
        }

        public void NotifyTurnEnd(BattleState state)
        {
            foreach (var p in _plugins.OfType<IBattlePlugin>()) p.OnTurnEnd(state);
        }

        public void NotifyBattleEnd(BattleState state, BattleResult result)
        {
            foreach (var p in _plugins.OfType<IBattlePlugin>()) p.OnBattleEnd(state, result);
        }

        public void NotifyFainted(BattleState state, BattlePokemon fainted)
        {
            foreach (var p in _plugins.OfType<IBattlePlugin>()) p.OnPokemonFainted(state, fainted);
        }

        public BattleState ApplyBeforeMove(BattleState state, BattleAction action)
            => _plugins.OfType<IBattlePlugin>()
                .Aggregate(state, (s, p) => p.OnBeforeMove(s, action) ?? s);

        public BattleState ApplyBeforeDamage(BattleState state, DamageResult damage)
            => _plugins.OfType<IBattlePlugin>()
                .Aggregate(state, (s, p) => p.OnBeforeDamage(s, damage) ?? s);
    }
    ```

    namespace SDK.Plugins.Randomizer;

    using SDK.Core.Interfaces;
    using SDK.Core.ValueObjects;

    public sealed class RandomizerPlugin : IBattlePlugin
    {
        private readonly Random _rng;
        private const int TypeCount = 18;

        public string Name => "Randomizer";

        public RandomizerPlugin(int seed) => _rng = new Random(seed);

        // Appeler avant RunBattle — _rng stateful, résultat dépend de l'ordre d'appel.
        public BattlePokemon RandomizePokemon(BattlePokemon p)
        {
            var newType1 = _rng.Next(1, TypeCount + 1);
            var raw2 = _rng.Next(0, TypeCount + 1);
            int? newType2 = (raw2 == 0 || raw2 == newType1) ? null : raw2;
            return p with { Type1Id = newType1, Type2Id = newType2 };
        }

        public void OnBattleStart(BattleState state) { }
        public void OnTurnStart(BattleState state) { }
        public void OnTurnEnd(BattleState state) { }
        public void OnBattleEnd(BattleState state, BattleResult result) { }
        public void OnPokemonFainted(BattleState state, BattlePokemon fainted) { }
        public void OnPokemonCaught(BattleState state, BattlePokemon caught, string zone) { }
        public void OnPokemonLevelUp(BattlePokemon pokemon, int oldLevel, int newLevel) { }
        public BattleState? OnBeforeMove(BattleState state, BattleAction action) => null;
        public BattleState? OnBeforeDamage(BattleState state, DamageResult damage) => null;
    }
